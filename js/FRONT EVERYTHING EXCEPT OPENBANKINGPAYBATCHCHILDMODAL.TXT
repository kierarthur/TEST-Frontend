
function deriveBankingPayBankCsvEvidence(dataLike) {
  const root = (dataLike && typeof dataLike === 'object') ? dataLike : {};
  const batch = (root.batch && typeof root.batch === 'object') ? root.batch : {};

  const src = (
    (root.bank_csv_export_json && typeof root.bank_csv_export_json === 'object') ? root.bank_csv_export_json
    : ((batch.bank_csv_export_json && typeof batch.bank_csv_export_json === 'object') ? batch.bank_csv_export_json : null)
  );

  const pick = (keys = []) => {
    for (const key of keys) {
      try {
        if (!src || !Object.prototype.hasOwnProperty.call(src, key)) continue;
        const v = src[key];
        if (v !== undefined && v !== null && String(v).trim() !== '') return v;
      } catch {}
    }
    return null;
  };

  const generatedAtUtc = String(
    pick(['generated_at_utc', 'csv_file_generated_at_utc'])
    || root.csv_file_generated_at_utc
    || batch.csv_file_generated_at_utc
    || ''
  ).trim() || null;

  return {
    raw: src,
    generated_at_utc: generatedAtUtc,
    generated_by_user_id: pick(['generated_by_user_id']),
    scope: pick(['scope']),
    filename: pick(['filename']),
    row_count: pick(['row_count']),
    total_amount: pick(['total_amount']),
    transfer_hash: pick(['transfer_hash']),
    provider_snapshot: pick(['provider_snapshot']) || batch.rail_provider_snapshot || batch.rail_provider || null,
    rail_env_snapshot: pick(['rail_env_snapshot']) || batch.rail_env_snapshot || null
  };
}

    const settlementDateSeedIso = String(
      batch?.authoritative_payment_date ||
      batchData?.authoritative_payment_date ||
      batchData?.payment_date ||
      batch?.pay_date ||
      ''
    ).trim();
    state.payment_date_uk = formatIsoToUkLocal(settlementDateSeedIso);
  }

  const goldenKeyAvailable = detectGoldenKeyAvailable();

  const askSuppressConfirm = async () => {
    const title = 'Suppress remittances for this settlement?';
    const msg = [
      'The batch will still be settled/executed if authorised.',
      'Payment state and reservations will still commit.',
      'Candidates/suppliers will not receive CloudTMS remittance advice for this settlement.',
      'You are responsible for any external communication.',
      '',
      'Continue?'
    ].join('\n');
    try {
      if (typeof window.openUIConfirm === 'function') {
        const out = await window.openUIConfirm({ title, message: msg, confirmText: 'Yes, suppress remittances', cancelText: 'No' });
        if (out && typeof out === 'object') return !!(out.confirmed || out.ok || out.yes);
        return !!out;
      }
    } catch {}
    try {
      if (typeof uiConfirmModal === 'function') {
        return await uiConfirmModal(title, msg, { kind: 'warn' });
      }
    } catch {}
    return !!window.confirm(`${title}\n\n${msg}`);
  };

  const render = () => {
    const dis = state.busy ? 'data-disabled="1" aria-disabled="true" style="opacity:.55;filter:saturate(0.6) brightness(0.9);"' : '';
    const mode = String(state.execution_mode || '').toUpperCase();
    const isStandard = mode === 'STANDARD_BANK';
    const isCsv = mode === 'CSV_SETTLEMENT';
    const isExternal = mode === 'EXTERNAL_SETTLEMENT';
    const isScheduled = isStandard && (String(state.schedule_kind || '').toUpperCase() === 'SCHEDULED');

    const selectedModeInfo = modeInfo[mode] || modeInfo.STANDARD_BANK;
    const disabledModeHint = selectedModeInfo && !selectedModeInfo.enabled
      ? `<div class="error" style="white-space:pre-wrap;">${enc(selectedModeInfo.disabledReason || 'This mode is unavailable.')}</div>`
      : '';

    const scheduleHint = (() => {
      if (!isStandard) return '<span class="mini" style="opacity:.85;">Scheduling is available for standard bank execution mode.</span>';
      if (!isScheduled) return '<span class="mini" style="opacity:.85;">Payment will be executed immediately.</span>';
      const iso = parseUkDateToIsoLocal(state.date_uk);
      const hm = parseTimeHm(state.time_uk);
      if (!iso || !hm) return '<span class="mini" style="opacity:.85;">Choose a UK date and time.</span>';
      const utcIso = ukLocalToUtcIso(iso, hm);
      if (!utcIso) return '<span class="mini" style="opacity:.85;">Unable to convert the selected date/time.</span>';
      return `<span class="mini" style="opacity:.85;">Scheduled (UK) → UTC: <span class="mono">${enc(utcIso)}</span></span>`;
    })();

    const effectiveDateSummary = (() => {
      if (isStandard) {
        if (!isScheduled) {
          const todayIso = londonTodayIsoLocal();
          return `<div class="mini" style="opacity:.9;">Immediate payment date: <span class="mono">${enc(todayIso ? formatIsoToUkLocal(todayIso) : 'today')}</span></div>`;
        }
        const iso = parseUkDateToIsoLocal(state.date_uk);
        return `<div class="mini" style="opacity:.9;">Scheduled payment date: <span class="mono">${enc(iso ? formatIsoToUkLocal(iso) : 'select a UK scheduled date')}</span></div>`;
      }
      const label = isCsv ? 'CSV settlement payment date' : (isExternal ? 'External settlement payment date' : 'Settlement payment date');
      const iso = parseUkDateToIsoLocal(state.payment_date_uk);
      return `<div class="mini" style="opacity:.9;">${enc(label)}: <span class="mono">${enc(iso ? formatIsoToUkLocal(iso) : 'select date')}</span></div>`;
    })();

    const authHint = (() => {
      if (!goldenKeyAvailable) return `<div class="mini" style="opacity:.85;">Normal authorisation rules apply (other authorisers may be required).</div>`;
      const m = String(state.auth_mode || '').trim().toUpperCase();
      if (m === 'GOLDEN_KEY') return `<div class="mini" style="opacity:.85;">🔑 Golden Key selected: this execution will be authorised immediately (no other authorisers required).</div>`;
      return `<div class="mini" style="opacity:.85;">Normal authorisation selected: other authorisers may be required.</div>`;
    })();

    return `
      <div id="${enc(rootId)}">
        <div class="card">
          <div class="row">
            <label>${enc(modalTitle)}</label>
            <div class="controls" style="display:flex;flex-direction:column;gap:10px;">
              ${state.err ? `<div class="error" style="white-space:pre-wrap;">${enc(state.err)}</div>` : ''}

              <div class="mini" style="opacity:.9;">${enc(modalIntro)}</div>

              <div class="card" style="padding:10px; display:flex; flex-direction:column; gap:10px;">
                <div style="display:flex;gap:10px;align-items:center;flex-wrap:wrap;">
                  <div class="mini" style="opacity:.85;">Batch</div>
                  <div class="mono" style="font-weight:700;">${enc(payBatchId ? `${payBatchId.slice(0, 8)}…` : '—')}</div>
                  <div class="mini" style="opacity:.85;">Scope</div>
                  <div class="mono">${enc(scope)}</div>
                </div>

                <div style="display:flex;flex-direction:column;gap:8px;">
                  ${['STANDARD_BANK','CSV_SETTLEMENT','EXTERNAL_SETTLEMENT'].map((k) => {
                    const mi = modeInfo[k];
                    const checked = mode === k ? 'checked' : '';
                    const disabled = mi.enabled ? '' : 'disabled';
                    return `<label class="inline mini" style="display:flex;gap:8px;align-items:flex-start; border:1px solid var(--line); border-radius:10px; padding:10px; ${mi.enabled ? '' : 'opacity:.65;'}">
                      <input type="radio" name="pay_exec_mode" value="${enc(k)}" ${checked} ${disabled} ${dis} data-action="pay-exec:setMode" />
                      <span>
                        <div style="font-weight:700;">${enc(mi.label)}</div>
                        <div style="opacity:.85;">${enc(mi.desc)}</div>
                        ${(!mi.enabled && mi.disabledReason) ? `<div class="mini" style="color:#b26b00;margin-top:4px;">${enc(mi.disabledReason)}</div>` : ''}
                        ${(k === 'CSV_SETTLEMENT' && csvGeneratedAtUtc) ? `<div class="mini" style="opacity:.75;margin-top:4px;">CSV generated: ${enc(csvGeneratedAtUtc)}</div>` : ''}
                      </span>
                    </label>`;
                  }).join('')}
                </div>
              </div>

              ${disabledModeHint}

              <div class="card" style="padding:10px; display:flex; flex-direction:column; gap:10px;">
                <div class="card" style="padding:8px;">
                  ${effectiveDateSummary}
                  ${
                    !isStandard
                      ? `
                        <div style="margin-top:8px;">
                          <label class="inline mini" style="opacity:.85;">${enc(isCsv ? 'CSV settlement payment date' : 'External settlement payment date')}</label>
                          <input id="payExecPaymentDateUk" class="input" type="text" value="${enc(state.payment_date_uk)}" placeholder="DD/MM/YYYY" autocomplete="off" spellcheck="false" data-uk-date="1" data-action="pay-exec:setPaymentDateUk" ${dis} />
                          <div class="mini" style="opacity:.75; margin-top:4px;">Required for ${enc(isCsv ? 'CSV settlement' : 'external settlement')}.</div>
                        </div>
                      `
                      : `<div class="mini" style="opacity:.75; margin-top:4px;">No separate payment-date field is shown for standard bank execution. The effective date is derived from the immediate/scheduled choice below.</div>`
                  }
                </div>

                ${isStandard ? `
                  ${!String(state.funding_account_ref || '').trim() ? `
                    <div class="notice warning mini" style="margin-bottom:8px;">
                      Funding account ref is missing. Standard bank execution remains available, but the backend will validate the funding account before submission.
                      <div style="margin-top:6px;">
                        <label class="inline mini" style="opacity:.85;">Funding account ref</label>
                        <input class="input" type="text" value="${enc(state.funding_account_ref)}" placeholder="Enter funding account ref if required" data-action="pay-exec:setFundingAccountRef" ${dis} />
                      </div>
                    </div>
                  ` : ''}
                  <div style="display:flex;gap:12px;align-items:center;flex-wrap:wrap;">
                    <label class="inline mini" style="display:flex;gap:6px;align-items:center;">
                      <input type="radio" name="pay_exec_schedule_kind" value="IMMEDIATE" ${String(state.schedule_kind).toUpperCase() === 'IMMEDIATE' ? 'checked' : ''} ${dis} data-action="pay-exec:setScheduleKind" />
                      Immediate
                    </label>
                    <label class="inline mini" style="display:flex;gap:6px;align-items:center;">
                      <input type="radio" name="pay_exec_schedule_kind" value="SCHEDULED" ${String(state.schedule_kind).toUpperCase() === 'SCHEDULED' ? 'checked' : ''} ${dis} data-action="pay-exec:setScheduleKind" />
                      Scheduled (UK)
                    </label>
                    <div style="margin-left:auto;">${scheduleHint}</div>
                  </div>
                  ${isScheduled ? `
                    <div style="display:flex;gap:10px;align-items:flex-end;flex-wrap:wrap;">
                      <div style="min-width:220px;">
                        <label class="inline mini" style="opacity:.85;">Date (UK)</label>
                        <input id="payExecDateUk" class="input" type="text" value="${enc(state.date_uk)}" placeholder="DD/MM/YYYY" data-uk-date="1" data-action="pay-exec:setDateUk" ${dis} />
                      </div>
                      <div style="min-width:180px;">
                        <label class="inline mini" style="opacity:.85;">Time (UK, HHMM)</label>
                        <input id="payExecTimeUk" class="input mono" type="text" value="${enc(state.time_uk)}" placeholder="e.g. 0930" maxlength="4" inputmode="numeric" data-action="pay-exec:setTimeUk" ${dis} />
                      </div>
                    </div>
                  ` : ''}

                  <details style="border:1px solid var(--line); border-radius:10px; padding:10px;">
                    <summary class="mini" style="cursor:pointer;opacity:.9;">Advanced ▸</summary>
                    <div style="margin-top:10px; display:flex; flex-direction:column; gap:10px;">
                      <div>
                        <label class="inline mini" style="opacity:.85;">Funding account ref</label>
                        <input id="payExecFundingAccountRef" class="input" type="text" value="${enc(state.funding_account_ref)}" placeholder="(optional)" data-action="pay-exec:setFundingAccountRef" ${dis} />
                      </div>
                      <div>
                        <label class="inline mini" style="opacity:.85;">Warning hours JSON</label>
                        <textarea id="payExecWarningHoursJson" class="input" rows="4" placeholder='(optional) JSON array, e.g. [{"name":"...","hours":12}]' data-action="pay-exec:setWarningHoursJson" ${dis} style="width:100%;">${enc(state.warning_hours_json_raw || '')}</textarea>
                      </div>
                    </div>
                  </details>
                ` : ''}

                ${isCsv ? `
                  <label class="inline mini" style="display:flex;gap:8px;align-items:flex-start;">
                    <input type="checkbox" ${state.csv_uploaded_confirmed ? 'checked' : ''} ${dis} data-action="pay-exec:setCsvUploadedConfirmed" />
                    <span>I confirm the CloudTMS CSV banking file has been uploaded/processed with the bank.</span>
                  </label>
                  <div>
                    <label class="inline mini" style="opacity:.85;">Bank confirmation reference</label>
                    <input class="input" type="text" value="${enc(state.csv_bank_confirm_ref)}" data-action="pay-exec:setCsvBankConfirmRef" placeholder="Required" ${dis} />
                  </div>
                ` : ''}

                ${isExternal ? `
                  <div>
                    <label class="inline mini" style="opacity:.85;">External settlement comment</label>
                    <textarea class="input" rows="3" data-action="pay-exec:setExternalComment" placeholder="Required" ${dis} style="width:100%;">${enc(state.external_settlement_comment)}</textarea>
                  </div>
                ` : ''}
              </div>

              <div class="card" style="padding:10px;">
                <label class="inline mini" style="display:flex;gap:8px;align-items:flex-start;">
                  <input id="payExecSuppressRemit" type="checkbox" ${state.suppress_remittances ? 'checked' : ''} ${dis} data-action="pay-exec:setSuppressRemittances" />
                  <span>Do not send remittances for this settlement</span>
                </label>
                <div class="mini" style="opacity:.75;margin-top:6px;">Default is off. If enabled, remittances are suppressed only for this execution action.</div>
              </div>

              <div class="card" style="padding:10px;">
                ${authHint}
                <div class="mini" style="opacity:.75;margin-top:8px;">Note: You will be asked to verify by password + email code once for this execution.</div>
                ${goldenKeyAvailable ? `<div style="margin-top:8px;"><label class="inline mini" style="display:flex;gap:6px;align-items:center;"><input type="radio" name="pay_exec_auth_mode" value="NORMAL" ${String(state.auth_mode).toUpperCase() === 'NORMAL' ? 'checked' : ''} ${dis} data-action="pay-exec:setAuthMode" />Normal authorisation</label> <label class="inline mini" style="display:flex;gap:6px;align-items:center;margin-top:6px;"><input type="radio" name="pay_exec_auth_mode" value="GOLDEN_KEY" ${String(state.auth_mode).toUpperCase() === 'GOLDEN_KEY' ? 'checked' : ''} ${dis} data-action="pay-exec:setAuthMode" />Use Golden Key</label></div>` : ''}
              </div>

              <div style="display:flex;gap:10px;align-items:center;flex-wrap:wrap;justify-content:flex-end;">
                <button id="payExecConfirmBtn" type="button" class="btn btn-primary" ${dis}>${state.busy ? 'Working…' : (confirmLabelOverride || 'Continue')}</button>
                <button id="payExecCancelBtn" type="button" class="btn btn-outline" ${dis}>Cancel</button>
              </div>
            </div>
          </div>
        </div>
      </div>
    `;
  };

  let ctxSeed = null;
  try {
    ctxSeed = { entity: 'banking-pay-exec-confirm', openToken: `banking-pay-exec-confirm:${Date.now()}:${Math.random().toString(36).slice(2)}` };
    window.modalCtx = ctxSeed;
    try { if (typeof modalCtx !== 'undefined') modalCtx = ctxSeed; } catch {}
  } catch {}

  return await new Promise((resolve) => {
    let done = false;
    let detachHandlers = null;

    const finish = (res) => { if (done) return; done = true; resolve(res || null); };
    const onDismiss = () => { try { if (typeof detachHandlers === 'function') detachHandlers(); } catch {} finish({ cancelled: true }); };
    const renderTab = (k) => (k === 'main' ? render() : '');

    showModal(modalTitle, [{ key: 'main', label: 'Confirm' }], renderTab, null, true, null, {
      kind, noParentGate: true, showSave: false, showApply: false, stayOpenOnSave: false, onDismiss
    });

    const wire = () => {
      const body = document.getElementById('modalBody');
      if (!body) return;
      const getRoot = () => { try { return body.querySelector(`#${CSS.escape(rootId)}`); } catch { return body.querySelector(`#${rootId}`); } };
      if (!getRoot()) return;
      if (body.__payExecConfirmWiredToken === rootId) return;
      body.__payExecConfirmWiredToken = rootId;

      const stillTopIsThisModal = () => {
        try {
          const fr = (typeof window.__getModalFrame === 'function') ? window.__getModalFrame() : null;
          if (!fr || String(fr.kind || '') !== String(kind || '')) return false;
          const ctx = (window.modalCtx && typeof window.modalCtx === 'object') ? window.modalCtx : null;
          if (!ctx || String(ctx.entity || '') !== 'banking-pay-exec-confirm') return false;
          if (ctxSeed && fr._ctxRef !== ctxSeed) return false;
          return true;
        } catch { return false; }
      };

      const ensureDatePicker = () => {
        try {
          const r = getRoot(); if (!r) return;
          const elDate = r.querySelector('#payExecDateUk');
          if (elDate && typeof attachUkDatePicker === 'function' && !elDate.__ukdp) {
            elDate.__ukdp = true;
            try { attachUkDatePicker(elDate, {}); } catch {}
          }
          const elPaymentDate = r.querySelector('#payExecPaymentDateUk');
          if (elPaymentDate && typeof attachUkDatePicker === 'function' && !elPaymentDate.__ukdp) {
            elPaymentDate.__ukdp = true;
            try { attachUkDatePicker(elPaymentDate, {}); } catch {}
          }
        } catch {}
      };

      const rerender = () => {
        try {
          const fr = (typeof window.__getModalFrame === 'function') ? window.__getModalFrame() : null;
          if (fr && typeof fr.setTab === 'function') {
            Promise.resolve(fr.setTab('main')).then(() => { try { ensureDatePicker(); } catch {} }).catch(() => {});
            return;
          }
        } catch {}
        try { body.innerHTML = render(); } catch {}
        try { ensureDatePicker(); } catch {}
      };

      const onChange = async (ev) => {
        try {
          if (!stillTopIsThisModal()) return;
          const t = ev && ev.target ? ev.target : null;
          const r = getRoot();
          if (!t || !r || !r.contains(t)) return;
          const act = String(t.getAttribute && t.getAttribute('data-action') || '').trim();

          if (act === 'pay-exec:setMode') {
            const v = String(t.value || '').trim().toUpperCase();
            if (modeInfo[v] && modeInfo[v].enabled) {
              state.execution_mode = v;
              state.err = '';
              rerender();
            }
            return;
          }

          if (act === 'pay-exec:setScheduleKind') {
            const v = String(t.value || '').trim().toUpperCase();
            state.schedule_kind = (v === 'SCHEDULED') ? 'SCHEDULED' : 'IMMEDIATE';
            state.err = '';
            rerender();
            return;
          }

          if (act === 'pay-exec:setAuthMode') {
            const v = String(t.value || '').trim().toUpperCase();
            state.auth_mode = (goldenKeyAvailable && v === 'GOLDEN_KEY') ? 'GOLDEN_KEY' : 'NORMAL';
            state.err = '';
            rerender();
            return;
          }

          if (act === 'pay-exec:setDateUk') {
            state.date_uk = String(t.value || '');
            state.err = '';
            rerender();
            return;
          }

          if (act === 'pay-exec:setPaymentDateUk') {
            state.payment_date_uk = String(t.value || '');
            state.err = '';
            return;
          }

          if (act === 'pay-exec:setSuppressRemittances') {
            const want = !!t.checked;
            if (!want) {
              state.suppress_remittances = false;
              state.suppress_remittances_confirmed = false;
              state.err = '';
              rerender();
              return;
            }
            const ok = await askSuppressConfirm();
            state.suppress_remittances = !!ok;
            state.suppress_remittances_confirmed = !!ok;
            state.err = '';
            rerender();
            return;
          }

          if (act === 'pay-exec:setCsvUploadedConfirmed') {
            state.csv_uploaded_confirmed = !!t.checked;
            state.err = '';
            return;
          }
        } catch {}
      };

      const onInput = (ev) => {
        try {
          if (!stillTopIsThisModal()) return;
          const t = ev && ev.target ? ev.target : null;
          const r = getRoot();
          if (!t || !r || !r.contains(t)) return;
          const act = String(t.getAttribute && t.getAttribute('data-action') || '').trim();

          if (act === 'pay-exec:setPaymentDateUk') { state.payment_date_uk = String(t.value || '').trim(); state.err = ''; return; }
          if (act === 'pay-exec:setDateUk') { state.date_uk = String(t.value || ''); state.err = ''; return; }
          if (act === 'pay-exec:setTimeUk') { const cleaned = sanitizeTimeDigits(String(t.value || '')); state.time_uk = cleaned; try { t.value = cleaned; } catch {} state.err = ''; return; }
          if (act === 'pay-exec:setFundingAccountRef') { state.funding_account_ref = String(t.value || ''); state.err = ''; return; }
          if (act === 'pay-exec:setWarningHoursJson') { state.warning_hours_json_raw = String(t.value || ''); state.err = ''; return; }
          if (act === 'pay-exec:setCsvBankConfirmRef') { state.csv_bank_confirm_ref = String(t.value || ''); state.err = ''; return; }
          if (act === 'pay-exec:setExternalComment') { state.external_settlement_comment = String(t.value || ''); state.err = ''; return; }
        } catch {}
      };

      const onClick = (ev) => {
        try {
          if (!stillTopIsThisModal()) return;
          const t = ev && ev.target ? ev.target : null;
          if (!t || typeof t.closest !== 'function') return;
          const r = getRoot();
          if (!r || !r.contains(t)) return;
          const btn = t.closest('button');
          if (!btn) return;
          const id = String(btn.id || '').trim();

          if (id === 'payExecCancelBtn') { finish({ cancelled: true }); closeTop(); return; }
          if (id !== 'payExecConfirmBtn') return;

          const mode = String(state.execution_mode || '').trim().toUpperCase();
          if (!modeInfo[mode] || !modeInfo[mode].enabled) {
            state.err = (modeInfo[mode] && modeInfo[mode].disabledReason) ? modeInfo[mode].disabledReason : 'Selected mode is unavailable.';
            rerender();
            return;
          }

          let payment_date = null;
          let schedule_kind = 'IMMEDIATE';
          let scheduled_at_utc = null;
          let funding_account_ref = null;
          let warning_hours_json = null;

          if (mode === 'STANDARD_BANK') {
            const skRaw = String(state.schedule_kind || '').trim().toUpperCase();
            schedule_kind = (skRaw === 'SCHEDULED') ? 'SCHEDULED' : 'IMMEDIATE';
            if (schedule_kind === 'SCHEDULED') {
              const iso = parseUkDateToIsoLocal(state.date_uk);
              const hm = parseTimeHm(state.time_uk);
              if (!iso) { state.err = 'Scheduled date is required (DD/MM/YYYY).'; rerender(); return; }
              if (!hm) { state.err = 'Scheduled time is required in 24-hour HHMM format (e.g. 0200, 2311).'; rerender(); return; }
              const utcIso = ukLocalToUtcIso(iso, hm);
              if (!utcIso) { state.err = 'Unable to convert the selected UK date/time.'; rerender(); return; }
              scheduled_at_utc = utcIso;
              payment_date = iso;
            } else {
              payment_date = londonTodayIsoLocal();
            }
            if (!/^\d{4}-\d{2}-\d{2}$/.test(String(payment_date || ''))) {
              state.err = 'Unable to resolve the effective payment date.';
              rerender();
              return;
            }
            funding_account_ref = String(state.funding_account_ref || '').trim() || null;
            const rawWarn = String(state.warning_hours_json_raw || '').trim();
            if (rawWarn) {
              try {
                const j = JSON.parse(rawWarn);
                if (!Array.isArray(j)) { state.err = 'Warning hours JSON must be a JSON array.'; rerender(); return; }
                warning_hours_json = j;
              } catch {
                state.err = 'Warning hours JSON is invalid JSON.';
                rerender();
                return;
              }
            }
          } else {
            payment_date = parseUkDateToIsoLocal(state.payment_date_uk);
            if (!payment_date) {
              state.err = mode === 'CSV_SETTLEMENT'
                ? 'CSV settlement payment date is required (DD/MM/YYYY).'
                : 'External settlement payment date is required (DD/MM/YYYY).';
              rerender();
              return;
            }
          }

          if (mode === 'CSV_SETTLEMENT') {
            if (!state.csv_uploaded_confirmed) { state.err = 'Confirm that the CloudTMS CSV banking file has been uploaded/processed with the bank.'; rerender(); return; }
            if (!String(state.csv_bank_confirm_ref || '').trim()) { state.err = 'Bank confirmation reference is required for CSV settlement.'; rerender(); return; }
          }

          if (mode === 'EXTERNAL_SETTLEMENT') {
            if (!String(state.external_settlement_comment || '').trim()) { state.err = 'External settlement comment is required.'; rerender(); return; }
          }

          const authMode = goldenKeyAvailable ? String(state.auth_mode || 'NORMAL').trim().toUpperCase() : 'NORMAL';
          const actor_intent = (authMode === 'GOLDEN_KEY') ? 'USE_GOLDEN_KEY' : null;

          finish({
            confirmed: true,
            execution_mode: mode,
            payment_date,
            schedule_kind,
            scheduled_at_utc,
            funding_account_ref: (mode === 'STANDARD_BANK') ? funding_account_ref : null,
            warning_hours_json,
            actor_intent,
            suppress_remittances: !!state.suppress_remittances,
            suppress_remittances_confirmed: !!(state.suppress_remittances && state.suppress_remittances_confirmed),
            csv_uploaded_confirmed: mode === 'CSV_SETTLEMENT' ? true : false,
            csv_bank_confirm_ref: mode === 'CSV_SETTLEMENT' ? String(state.csv_bank_confirm_ref || '').trim() : null,
            external_settlement_comment: mode === 'EXTERNAL_SETTLEMENT' ? String(state.external_settlement_comment || '').trim() : null,
            pay_channel_scope: scope,
            scope
          });
          closeTop();
        } catch {}
      };

      body.addEventListener('change', onChange, true);
      body.addEventListener('input', onInput, true);
      body.addEventListener('click', onClick, true);
      detachHandlers = () => {
        try { body.removeEventListener('change', onChange, true); } catch {}
        try { body.removeEventListener('input', onInput, true); } catch {}
        try { body.removeEventListener('click', onClick, true); } catch {}
        try { if (body.__payExecConfirmWiredToken === rootId) body.__payExecConfirmWiredToken = null; } catch {}
      };

      ensureDatePicker();
    };

    try { requestAnimationFrame(() => requestAnimationFrame(wire)); } catch { setTimeout(wire, 0); }
  });

              <div style="display:flex;gap:8px;align-items:center;flex-wrap:wrap;">
                ${primaryActionButtonsHtml}
                ${timesheetSummaryButtonHtml}
                  <button type="button" class="btn btn-sm btn-outline" data-action="banking:pay:child:refresh">Refresh Batch</button>
                <button type="button" class="btn btn-sm btn-outline" data-action="banking:pay:child:exportDetailCsv">Detailed CSV</button>
                <button type="button" class="btn btn-sm btn-outline" data-action="banking:pay:child:exportDetailPdf">Detailed PDF</button>
                ${busyLine}
              </div>
              <div class="mini" style="opacity:.88;">Remittances sent: <span class="mono">${enc(remittancesSent ? 'Yes' : 'No')}</span></div>
              <div class="mini" style="opacity:.82;">${enc(isTrueEmptyShell ? emptyShellMessage : (executionRefreshApplied ? 'Batch overview and payment status have been refreshed after payment execution. Detail tabs still load paged rows on demand.' : 'This batch is in bootstrap mode. Open the relevant detail tabs to load paged rows.'))}</div>
            </div>
          </div>
        </div>
      </div>
    `;
  }

  const candidateRowsHtml = overviewRowsHtml
    ? overviewRowsHtml
    : (candidateViewModels.length
      ? `
      <div style="overflow:auto; border:1px solid var(--line); border-radius:10px; max-height:min(60vh,820px);">
        <table class="grid" style="min-width:1160px; table-layout:fixed; margin:0;">
          <thead>
            <tr>
              <th style="width:56px; position:sticky; top:0; z-index:1; background:var(--panel,#fff);">
                <button
                  type="button"
                  class="btn btn-sm btn-outline"
                  data-action="${allExpanded ? 'banking:pay:child:collapseAll' : 'banking:pay:child:expandAll'}"
                  title="${allExpanded ? 'Collapse all' : 'Expand all'}"
                  style="min-width:32px; padding:0 8px;"
                >${allExpanded ? '−' : '+'}</button>
              </th>
              <th style="position:sticky; top:0; z-index:1; background:var(--panel,#fff);">Candidate name / TMS ref</th>
              <th style="position:sticky; top:0; z-index:1; background:var(--panel,#fff);">Payee / Umbrella</th>
              <th style="width:120px;text-align:right;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Gross</th>
              <th style="width:120px;text-align:right;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Overpayment</th>
              <th style="width:180px;text-align:right;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Payment Advance Repayment</th>
              <th style="width:140px;text-align:right;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Final payable</th>
              <th style="width:180px;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Status</th>
            </tr>
          </thead>
          <tbody>
            ${candidateViewModels.map((row) => {
              const c = row.candidate;
              const candidateId = row.candidateId;
              const candidateName = getCandidateName(c);
              const tmsRef = getCandidateTmsRef(c);
              const payeeName = getPayeeName(c);
              const expansionKey = getCandidateExpansionKey(c) || candidateId;
              const expandedCandidate = isCandidateExpanded(c);
              const lineRows = expandedCandidate ? buildBreakdownRowsForCandidate(c) : [];
              const statusInfo = row.statusInfo;
              const detailHtml = expandedCandidate
                ? `
                  <tr>
                    <td colspan="8" style="padding:0;">
                      <div style="padding:10px 12px;background:rgba(0,0,0,.02);border-top:1px solid var(--line);">
                        <div style="display:flex;gap:8px;align-items:center;flex-wrap:wrap;margin-bottom:8px;">
                          <div class="mini" style="font-weight:700;opacity:.9;">Line items</div>
                          <div class="mini" style="opacity:.85;">${enc(String(lineRows.length))} row${lineRows.length === 1 ? '' : 's'}</div>
                        </div>
                        ${lineRows.length ? `
                          <div style="max-height:min(42vh,460px);overflow:auto;border:1px solid var(--line);border-radius:10px;">
                            <table class="grid" style="min-width:1020px;table-layout:auto;margin:0;">
                              <thead>
                                <tr>
                                  <th style="width:130px;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Week ending</th>
                                  <th style="position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Client</th>
                                  <th style="width:170px;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Unit/category</th>
                                  <th style="width:170px;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Line type</th>
                                  <th style="width:90px;text-align:right;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Qty</th>
                                  <th style="width:110px;text-align:right;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Rate</th>
                                  <th style="width:120px;text-align:right;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Amount</th>
                                  <th style="width:180px;position:sticky;top:0;z-index:1;background:var(--panel,#fff);">Destination</th>
                                </tr>
                              </thead>
                              <tbody>
                                ${lineRows.map((line) => `
                                  <tr>
                                    <td class="mini" style="white-space:nowrap;">${enc(line.week_ending_date ? formatIsoToUkLocal(line.week_ending_date) : '—')}</td>
                                    <td class="mini">${enc(line.client_name || '—')}</td>
                                    <td class="mini">${enc(line.unit_or_category_text || '—')}</td>
                                    <td class="mini">${enc(line.line_type_text || '—')}</td>
                                    <td class="mono" style="text-align:right;">${enc(line.qty_text || '—')}</td>
                                    <td class="mono" style="text-align:right;">${enc(line.rate_text ? `£${line.rate_text}` : '—')}</td>
                                    <td class="mono" style="text-align:right;">£${enc(line.amount_text || '0.00')}</td>
                                    <td class="mini">${enc(line.destination_text || '—')}</td>
                                  </tr>
                                `).join('')}
                              </tbody>
                            </table>
                          </div>
                        ` : `<div class="mini" style="opacity:.85;">No line items were returned for this candidate.</div>`}
                      </div>
                    </td>
                  </tr>
                `
                : '';
              return `
                <tr>
                  <td style="width:56px;">
                    <button
                      type="button"
                      class="btn btn-sm btn-outline"
                      data-action="banking:pay:child:toggleExpandCandidate"
                      data-candidate-id="${enc(expansionKey)}"
                      data-pay-batch-candidate-id="${enc(trimStr(c?.pay_batch_candidate_id || c?.payBatchCandidateId || ''))}"
                      data-candidate-row-id="${enc(trimStr(c?.id || c?.candidate_row_id || c?.candidateRowId || ''))}"
                      data-actual-candidate-id="${enc(candidateId)}"
                      title="${expandedCandidate ? 'Collapse' : 'Expand'}"
                      style="min-width:32px;padding:0 8px;"
                    >${expandedCandidate ? '−' : '+'}</button>
                  </td>
                  <td>
                    <div style="display:flex;flex-direction:column;gap:4px;">
                      <div style="font-weight:700;">${enc(candidateName)}</div>
                      ${tmsRef ? `<div class="mono mini">${enc(tmsRef)}</div>` : ''}
                    </div>
                  </td>
                  <td class="mini">${enc(payeeName)}</td>
                  <td class="mono" style="text-align:right;">£${enc(fmtMoney(row.financials.gross))}</td>
                  <td class="mono" style="text-align:right;">£${enc(fmtMoney(row.financials.overpayment))}</td>
                  <td class="mono" style="text-align:right;">£${enc(fmtMoney(row.financials.loan))}</td>
                  <td class="mono" style="text-align:right;">£${enc(fmtMoney(row.financials.finalPayable))}</td>
                  <td>
                    <span class="${enc(statusInfo.tone === 'sent' ? 'pill pill-ok' : statusInfo.tone === 'failed' ? 'pill pill-warn' : 'pill pill-info')}">${enc(`${statusInfo.symbol} ${statusInfo.label}`)}</span>
                  </td>
                </tr>
                ${detailHtml}
              `;
            }).join('')}
          </tbody>
        </table>
      </div>
    `
      : `<div class="mini" style="opacity:.85;">${overviewItemsLoading ? 'Loading overview rows…' : (isTrueEmptyShell ? enc(emptyShellMessage) : 'No beneficiaries found in this batch.')}</div>`);

  return `
    <div id="${enc(rootId)}">
      <div class="card">
        <div class="row">
          <label>Overview</label>
          <div class="controls" style="display:flex;flex-direction:column;gap:10px;">
            ${errHtml}
            <div style="display:flex;gap:10px;align-items:center;flex-wrap:wrap;">
              <span class="mini" style="font-weight:800;">${enc(providerEnvHeader)}</span>
              <span class="mini">Pay date: <span class="mono">${enc(payDate ? formatIsoToUkLocal(payDate) : '—')}</span></span>
              <div style="margin-left:auto;display:flex;gap:8px;align-items:center;flex-wrap:wrap;">
                <button type="button" class="btn btn-sm btn-outline" data-action="banking:pay:child:refresh" title="Refresh batch">Refresh Batch</button>
                ${busyLine}
              </div>
            </div>

            ${batchOutcomeHtml}
            ${postSettlementCommunicationWarningHtml}
            ${providerSubmitOverviewHtml}
            ${webhookEvidenceHtml}
            ${liveProgressHtml}
            ${retryCardHtml}
            ${activeOperationSummaryHtml}
            ${renderPaymentSummaryHtml()}

            <div class="card" style="padding:10px;">
              <div class="mini" style="font-weight:700;opacity:.9;margin-bottom:8px;">Actions</div>
              <div style="display:flex;gap:8px;align-items:center;flex-wrap:wrap;">
                ${primaryActionButtonsHtml}
                ${timesheetSummaryButtonHtml}
                <button type="button" class="btn btn-sm btn-outline" data-action="banking:pay:child:refresh" title="Refresh batch">Refresh Batch</button>
                <button type="button" class="btn btn-sm btn-outline" data-action="banking:pay:child:exportDetailCsv" title="Download detailed CSV">Detailed CSV</button>
                <button type="button" class="btn btn-sm btn-outline" data-action="banking:pay:child:exportDetailPdf" title="Download detailed PDF">Detailed PDF</button>
              </div>
            </div>

            <div class="mini" style="opacity:.88;">Remittances sent: <span class="mono">${enc(remittancesSent ? 'Yes' : 'No')}</span></div>

            <div class="card" style="padding:10px;">
              <div class="mini" style="font-weight:700;opacity:.9;margin-bottom:8px;">Beneficiaries</div>
              ${candidateRowsHtml}
            </div>
          </div>
        </div>
      </div>
    </div>
  `;
}
